Function Description Example
count() Get array size count([1,2,3]) = 3
array_push() Add to end array_push($arr, $val)
array_pop() Remove from end $last = array_pop($arr)
array_shift() Remove from start $first = array_shift($arr)
array_unshift() Add to start array_unshift($arr, $val)
sort() Sort ascending sort($arr)
rsort() Sort descending rsort($arr)
array_filter() Keep matching array_filter($arr, callback)
array_map() Transform each array_map(callback, $arr)
array_merge() Combine arrays array_merge($arr1, $arr2)
array_slice() Extract portion array_slice($arr, $start, $len)
array_search() Find value $key = array_search($val, $arr)
in_array() Check if exists in_array($val, $arr)
array_column() Extract column array_column($data, 'id')
array_unique() Remove duplicates array_unique($arr)
array_reverse() Flip order array_reverse($arr)
array_keys() Get all keys array_keys($arr)
array_values() Get all values array_values($arr)
array_sum() Sum elements array_sum([1,2,3]) = 6
max()/min() Get extreme value max($arr), min($arr)

Function Description Example
strlen() Get string length strlen("hello") = 5
strtoupper() Convert to uppercase strtoupper("hello") = "HELLO"
strtolower() Convert to lowercase strtolower("HELLO") = "hello"
ucfirst() Capitalize first letter ucfirst("hello") = "Hello"
ucwords() Capitalize each word ucwords("hello world") = "Hello World"
str_replace() Replace text (all) str_replace("a", "b", "apple") = "bpple"
str_ireplace() Replace (case-insensitive) str_ireplace("HELLO", "hi", "Hello")
substr() Get substring substr("hello", 0, 2) = "he"
strpos() Find first position strpos("hello", "l") = 2
strrpos() Find last position strrpos("hello", "l") = 3
trim() Remove whitespace trim(" hello ") = "hello"
ltrim()/rtrim() Remove left/right spaces ltrim(" x"), rtrim("x ")
explode() Split by delimiter explode(",", "a,b,c") = ["a","b","c"]
implode() Join array implode(",", ["a","b"]) = "a,b"
str_pad() Pad string str_pad("x", 5, "0", STR_PAD_LEFT)
str_repeat() Repeat string str_repeat("*", 5) = "*****"
strrev() Reverse string strrev("hello") = "olleh"
str_contains() Check if contains (PHP 8) str_contains("hello", "ll")
str_starts_with() Check start (PHP 8) str_starts_with("hello", "he")
str_ends_with() Check end (PHP 8) str_ends_with("hello", "lo")
